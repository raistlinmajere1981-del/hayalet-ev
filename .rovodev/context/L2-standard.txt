  1: /**
  2:  * IPC Client - Type-Safe IPC Wrapper
  3:  * 
  4:  * Bu modül renderer process'ten main process'e type-safe IPC çağrıları yapmayı sağlar.
  5:  * 
  6:  * @example
  7:  * import { IPC } from "./ipc-client.js";
  8:  * 
  9:  * // Type-safe çağrı - params ve result tipleri otomatik
 10:  * const result = await IPC.invoke("db-get-conversations", { accountId: "123" });
 11:  * if (result.success) {
 12:  *   console.log(result.data); // DbConversation[] tipinde
 13:  * }
 14:  * 
 15:  * // Hata durumunda null döner
 16:  * const safeResult = await IPC.safeInvoke("db-get-messages", { accountId: "123", conversationId: "456" });
 17:  */
 18: 
 19: import { LogManager } from "./log-manager.js";
 20: import { getErrorMessage } from "../types/index.js";
 21: import type { IPCChannel, IPCParams, IPCResult } from "../types/ipc-channels.js";
 22: 
 23: // ============================================================================
 24: // Types
 25: // ============================================================================
 26: 
 27: interface ElectronAPI {
 28:   invoke: (channel: string, params?: unknown) => Promise<unknown>;
 29:   send: (channel: string, params?: unknown) => void;
 30:   on: (channel: string, callback: (...args: unknown[]) => void) => () => void;
 31: }
 32: 
 33: // ============================================================================
 34: // IPC Client Class
 35: // ============================================================================
 36: 
 37: class IPCClientClass {
 38:   private _api: ElectronAPI | null = null;
 39: 
 40:   /**
 41:    * ElectronAPI referansını al (lazy init)
 42:    */
 43:   private get api(): ElectronAPI {
 44:     if (!this._api) {
 45:       const electronAPI = (window as unknown as { electronAPI?: ElectronAPI }).electronAPI;
 46:       if (!electronAPI) {
 47:         throw new Error("electronAPI not available - preload script may not have loaded");
 48:       }
 49:       this._api = electronAPI;
 50:     }
 51:     return this._api;
 52:   }
 53: 
 54:   /**
 55:    * Type-safe IPC invoke
 56:    * 
 57:    * @param channel - IPC kanal adı
 58:    * @param params - Kanal parametreleri (tip otomatik)
 59:    * @returns Promise<Result> - Kanal sonucu (tip otomatik)
 60:    * @throws Error - IPC hatası durumunda
 61:    * 
 62:    * @example
 63:    * const result = await IPC.invoke("db-get-conversations", { accountId: "123" });
 64:    */
 65:   async invoke<T extends IPCChannel>(
 66:     channel: T,
 67:     ...args: IPCParams<T> extends void ? [] : [IPCParams<T>]
 68:   ): Promise<IPCResult<T>> {
 69:     const params = args[0];
 70:     return this.api.invoke(channel, params) as Promise<IPCResult<T>>;
 71:   }
 72: 
 73:   /**
 74:    * Safe IPC invoke - hata durumunda null döner, exception fırlatmaz
 75:    * 
 76:    * @param channel - IPC kanal adı
 77:    * @param params - Kanal parametreleri
 78:    * @returns Promise<Result | null> - Sonuç veya null
 79:    * 
 80:    * @example
 81:    * const result = await IPC.safeInvoke("db-get-messages", { accountId: "123", conversationId: "456" });
 82:    * if (result) {
 83:    *   // result tipi otomatik
 84:    * }
 85:    */
 86:   async safeInvoke<T extends IPCChannel>(
 87:     channel: T,
 88:     ...args: IPCParams<T> extends void ? [] : [IPCParams<T>]
 89:   ): Promise<IPCResult<T> | null> {
 90:     try {
 91:       const params = args[0];
 92:       return await this.api.invoke(channel, params) as IPCResult<T>;
 93:     } catch (error) {
 94:       LogManager.addEvent("ipc-client", "error", `IPC invoke failed: ${channel} - ${getErrorMessage(error)}`);
 95:       return null;
 96:     }
 97:   }
 98: 
 99:   /**
100:    * IPC send (fire-and-forget)
101:    * 
102:    * @param channel - IPC kanal adı
103:    * @param params - Kanal parametreleri
104:    */
105:   send<T extends IPCChannel>(
106:     channel: T,
107:     ...args: IPCParams<T> extends void ? [] : [IPCParams<T>]
108:   ): void {
109:     const params = args[0];
110:     this.api.send(channel, params);
111:   }
112: 
113:   /**
114:    * IPC event listener
115:    * 
116:    * @param channel - IPC kanal adı
117:    * @param callback - Event callback
118:    * @returns Unsubscribe fonksiyonu
119:    */
120:   on<T extends IPCChannel>(
121:     channel: T,
122:     callback: (result: IPCResult<T>) => void
123:   ): () => void {
124:     return this.api.on(channel, (data) => callback(data as IPCResult<T>));
125:   }
126: 
127:   /**
128:    * ElectronAPI mevcut mu kontrol et
129:    */
130:   isAvailable(): boolean {
131:     return !!(window as unknown as { electronAPI?: ElectronAPI }).electronAPI;
132:   }
133: }
134: 
135: /**
136:  * Singleton IPC Client instance
137:  */
138: export const IPC = new IPCClientClass();

488:     wrapper._eventType = eventType;
489:     this._listeners.push(wrapper);
490:     return () => {

569:    * State transition with validation
570:    * @returns TransitionResult with success status and details
571:    */
572:   _transition(slot: string, newState: string, reason?: string): TransitionResult {
573:     const slotState = this._slots[slot];
574:     const oldState = slotState.state;
575:     
576:     // State machine ile geçerlilik kontrolü ve geçiş
577:     const result = slotState.stateMachine.transition(
578:       newState as SlotStateType,
579:       {
580:         correlationId: slotState.correlationId || undefined,
581:         reason,
582:       }
583:     );
584:     
585:     if (result.success) {
586:       // Legacy state alanını da güncelle (backward compatibility)
587:       slotState.state = newState;
588:       slotState.lastActivity = Date.now();
589:       
590:       if (!result.wasNoOp) {
591:         this._emit(slot, SlotEvent.STATE_CHANGED, { 
592:           oldState, 
593:           newState,
594:           validated: true,
595:         });
596:       }
597:     } else {
598:       // Geçersiz transition - hata logla ama crash etme
599:       this._log(slot, "error", `Invalid state transition blocked: ${oldState} → ${newState}. ${result.error}`);
600:     }
601:     
602:     return result;
603:   }
